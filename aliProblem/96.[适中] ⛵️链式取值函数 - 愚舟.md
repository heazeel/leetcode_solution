```
const data = {
  a: {
    aa: {
      aaa: 'raaa'
    }
  },
  b: [{}, {}, {bb: 'rbb'}]
}

getData(data, 'a.aa.aaa') // raaa
getData(data, 'b.2.bb') // rbb
getData(data, 'c.cc.ccc') // undefined
```
