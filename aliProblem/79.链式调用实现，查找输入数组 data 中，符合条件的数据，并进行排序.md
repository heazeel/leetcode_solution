链式调用实现，查找输入数组 data 中，符合条件的数据，并进行排序

#### js代码如下
```js
const data = [
  {userId: 8,  title: 'title1'},
  {userId: 11, title: 'other'},
  {userId: 15, title: null},
  {userId: 19, title: 'title2'}
];
const find = function(data) {
  // your code are here...
}
// 测试用例
// where 函数传入一个对象 key 为要查找的字段，value 为正则表达式
// orderBy 函数第一个参数为要排序的 key 第二个参数为 asc: 正序  desc: 倒序
const result = find(data).where({
  'title': /\d$/
}).orderBy('userId', 'desc');
console.log(result);// [{ userId: 19, title: 'title2'}, { userId: 8, title: 'title1' }];


const result = find(data).where({
  'title': /^title.*/
}).orderBy('userId', 'asc');
console.log(result);// [{userId: 8,  title: 'title1'}, {userId: 19, title: 'title2'}];

const result = find(data).where({
  'name': /^title.*/
})
console.log(result); // [];
```
