#### js代码如下
```js

const startTime = new Date().valueOf();

function getFakeRequest(s, i) {
  return () => new Promise(resolve => {
    setTimeout(() => resolve(), s * 1000);
  }).then(() => {
    const responseTime = Math.round((new Date().valueOf() - startTime) / 1000);
    // 距离一开始执行程序时的时间
    console.log(`第${i + 1}个请求，在${responseTime}s后响应`);
  });
}

// 题目要求：假设同时可以有 n 个请求，要求在尽可能的短的时间内完成所有请求怎么办？
function getBatchFetch(n) {
  // coding...
  const queue = [];
  // 可用带宽
  let availableBuffer = n;
  const fetch = () => {
    // 先判断是否有可用带宽，且有未发出的请求
    if (availableBuffer <= n && availableBuffer > 0 && queue.length) {
      // 消耗带宽-1
      availableBuffer--;
      // 发起请求
      queue
        .shift()()
        .then(() => {
          // 请求响应恢复带宽，并重新发起下个请求
          availableBuffer++;
          fetch();
        });
    }
  };
  return req => {
    // 推进堆栈，发起请求
    queue.push(req);
    fetch();
  };
}

// 假设最多可以并发 3 个请求
const batchFetch = getBatchFetch(3);

[1, 2, 3, 4, 4, 4].map(getFakeRequest).forEach(batchFetch);
// 预期输出结果
// 第1个请求，在1s后响应
// 第2个请求，在2s后响应
// 第3个请求，在3s后响应
// 第4个请求，在5s后响应
// 第5个请求，在6s后响应
// 第6个请求，在7s后响应
```
