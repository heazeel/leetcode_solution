实现一个函数 deepMerge，它可以深度合并两个对象，并处理对象的嵌套、数组的合并等。其中：

1、如果同一个键在两个对象中都存在且值都是对象，则进行递归合并。
2、如果同一个键在两个对象中都存在且值都是数组，则进行连接操作。
3、其他情况下，以第二个对象（参数）中的值为准。

```
function deepMerge(obj1, obj2) {
    // 实现你的逻辑
}

// 示例：
const obj1 = {
    a: 1,
    b: {
        b1: [1, 2, 3],
        b2: "hello"
    },
    c: [1, 2]
};

const obj2 = {
    b: {
        b1: [4, 5],
        b2: {
            b21: "world"
        }
    },
    c: [3, 4],
    d: 4
};

console.log(deepMerge(obj1, obj2));
// 输出结果应为：
{
     a: 1,
     b: {
         b1: [1, 2, 3, 4, 5],
         b2: {
             b21: "world"
         }
    },
    c: [1, 2, 3, 4],
    d: 4
}

```

#### js 代码如下

```js
function deepMerge(obj1, obj2) {
  let res = {};
  // 实现你的逻辑
  Object.keys(obj1).forEach((key) => {
    if (obj2.hasOwnProperty(key)) {
      if (
        obj1[key] !== null &&
        typeof obj1[key] === "object" &&
        !Array.isArray(obj1[key])
      ) {
        res[key] = deepMerge(obj1[key], obj2[key]);
      } else if (Array.isArray(obj1[key]) && Array.isArray(obj2[key])) {
        res[key] = obj1[key].concat(obj2[key]);
      } else {
        res[key] = obj2[key];
      }
    } else {
      res[key] = obj1[key];
    }
  });

  Object.keys(obj2).forEach((key) => {
    if (obj2.hasOwnProperty(key) && !res.hasOwnProperty(key)) {
      res[key] = obj2[key];
    }
  });

  return res;
}

// 示例：
const obj1 = {
  a: 1,
  b: {
    b1: [1, 2, 3],
    b2: "hello",
  },
  c: [1, 2],
};

const obj2 = {
  b: {
    b1: [4, 5],
    b2: {
      b21: "world",
    },
  },
  c: [3, 4],
  d: 4,
};

console.log(JSON.stringify(deepMerge(obj1, obj2)));
// 输出结果应为：
// {
//      a: 1,
//      b: {
//          b1: [1, 2, 3, 4, 5],
//          b2: {
//              b21: "world"
//          }
//     },
//     c: [1, 2, 3, 4],
//     d: 4
// }
```
