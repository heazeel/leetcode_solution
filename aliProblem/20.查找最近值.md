找到数组中最接近的值

#### js 代码如下

```js
/**
	给定一个数组Array，在数组中找到最接近值
  findCloseNum(Array,Num);
  要求输出 结果 ：12
*/

const Array = [1, 3, 54, 3, 65456, 345323, 12];
const Num = 14;

const findCloseNum = (arr, num) => {
  // 取相近值
  return arr.reduce((pre, cur) => {
    return Math.abs(pre - Num) > Math.abs(cur - Num) ? cur : pre;
  });
};
```
