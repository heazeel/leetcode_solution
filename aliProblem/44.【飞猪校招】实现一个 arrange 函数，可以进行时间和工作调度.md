> 实现一个arrange函数，可以进行时间和工作调度

```
/**
 * 实现一个arrange函数，可以进行时间和工作调度
 */
function arrange() {
  // your code ...   
}

// 用例参考
arrange('William');
// 输出： William is notified

arrange('William').wait(5).do('commit');
// 输出：William is notified
// 等待 5 秒
// 输出：Start to commit

arrange('William').waitFirst(5).do('push');
// 等待 5 秒
// 输出： Start to push
// 输出： William is notified
```
