> 手动实现节流

```js
const throttle = (func, delay) => {
  // 你的代码
  let lastTime = 0;
  return function (...args) {
    const now = Date.now();
    if (now - lastTime >= delay) {
      lastTime = now;
      func(...args);
    }
  };
};
const sayHi = (action) => console.log(`节流${action}`);
const onThrottle = throttle(sayHi, 1000);

window.onscroll = () => onThrottle("执行了");
```
