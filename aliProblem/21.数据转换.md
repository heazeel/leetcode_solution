#### 实现一个数据转化

> 将给定的目标数组转换为最终结果

```
// 示例结果：
const list = [
  {id: 1, name: '部门1', pid: 0},
  {id: 2, name: '部门1-1', pid: 1},
  {id: 3, name: '部门1-1-1', pid: 2},
  {id: 4, name: '部门2', pid: 0},
];
```

#### js 代码如下

```js
/** 
输出：
[
  {id: 1, name: '部门1', pid: 0},
  {id: 2, name: '部门1-1', pid: 1},
  {id: 3, name: '部门1-1-1', pid: 2},
  {id: 4, name: '部门2', pid: 0},
];
*/

const initiale = [
  {
    id: 1,
    name: "部门1",
    children: [
      {
        id: 2,
        name: "部门1-1",
        children: [
          {
            id: 3,
            name: "部门1-1-1",
            children: [],
          },
        ],
      },
    ],
  },
  {
    id: 4,
    name: "部门2",
    children: [],
  },
];

function treeTransList(tree) {
  let res = [];

  const dfs = (children, pid) => {
    if (children.length) {
      children.forEach((item) => {
        let obj = { id: item.id, name: item.name, pid };
        res.push(obj);

        dfs(item.children, item.id);
      });
    }
  };

  dfs(tree, 0);

  return res;
}

treeTransList(initiale, "children");
```
