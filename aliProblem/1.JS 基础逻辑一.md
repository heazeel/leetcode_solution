题目：找出数组中的连续子数组，使得这些子数组的和等于给定的目标值

> 给定一个整数数组 arr 和一个目标值 target，请编写一个函数 findSubArrays 找出所有和等于 target 的连续子数组。例如：

```
function findSubArrays(arr, target) {
    // 在这里编写你的解决方案
}

// 示例
console.log(findSubArrays([1, 2, 3, 4, 5], 5)); // 输出: [[2, 3], [5]]
console.log(findSubArrays([1, -1, 2, 3, -2, 4, 1], 3)); // 输出: [[1, -1, 2, 1], [3], [2, 3, -2]]
console.log(findSubArrays([1, 2, 3], 7)); // 输出: []
```

#### js 代码如下

```js
function findSubArrays(arr, target) {
  // 在这里编写你的解决方案
  let res = [];
  // 可补充
  for (let i = 0; i < arr.length; i++) {
    let sum = 0;
    let curArr = [];
    for (let j = i; j < arr.length; j++) {
      sum += arr[j];
      curArr.push(arr[j]);
      if (sum === target) {
        res.push([...curArr]);
      }
    }
  }

  return res;
}

// 示例
console.log(findSubArrays([1, 2, 3, 4, 5], 5)); // 输出: [[2, 3], [5]]
console.log(findSubArrays([1, -1, 2, 3, -2, 4, 1], 3)); // 输出: [[1, -1, 2, 1], [3], [2, 3, -2]]
console.log(findSubArrays([1, 2, 3], 7)); // 输出: []
```
