> 实现函数 sleep，入参为 毫秒，作用是停止执行制定时间。

#### js 代码如下

```js
function sleep(time) {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve(true);
    }, time);
  });
}

const run = async () => {
  // 第一步 执行 a = 1;
  const a = 1;
  console.log(a);

  // 停止 3 秒钟，再执行 第二步
  await sleep(3000);

  // 第二步 执行 b = 2;
  const b = 2;
  console.log(b);
};
```
