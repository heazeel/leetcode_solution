#### js代码如下
```js
// 存在以下异步回调请求方法，请将其封装成返回 Promise 的函数
function request(url, cb) {
  // 无需实现，这是一个已存在的函数
}

// 该函数调用示例
request('https://www.example.com', (error, resp) => {
  console.log('error', error)
  console.log('resp', resp)
})

// 1. 请将其封装成返回 Promise 的函数
/**
 * @returns {Promise<any>}
 */
function requestAsync(url) {
  // 请完成
}

// 函数调用示例
const resp = await requestAsync('https://www.example.com');

// 输出结果：
// 预期 resp 为
// {
//     "success": true,
//     "data": {
//         "url": "https://www.example.com"
//     }
// }


// 2 并增加超时 - 加分项
/**
 * @returns {Promise<any>}
 */
function requestAsyncWithTimeout(url, { timeout } = {}) {
  // 请完成
}

// 函数调用示例
const resp = await requestAsync('https://www.example.com', { timeout: 500 });
```
