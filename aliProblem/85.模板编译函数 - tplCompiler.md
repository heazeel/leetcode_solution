> 实现模板占位数据替换：
> 如：

```javascript
const tpl = '<p>origin：${protocol}://${hostname}:${port}</p>';`

const model = {
  protocol: 'https',
  hostname: 'foo.com',
  port: '8080'
};

function tplCompiler(tpl, model){
	// return '<p>origin：https://foo.com:8080</p>';
}
```
